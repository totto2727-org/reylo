func reyloOracleValue(value reflect.Value) any {
 if !value.IsValid() {return nil}
 if value.Kind()==reflect.Interface {if value.IsNil(){return nil};return reyloOracleValue(value.Elem())}
 if value.CanInterface() {
  switch v:=value.Interface().(type) {
  case cty.Value:
   if v==cty.NilVal{return map[string]any{"cty":"nil"}}
   if !v.IsKnown(){return map[string]any{"cty":"unknown"}}
   if v.IsNull(){return map[string]any{"cty":"null"}}
   switch v.Type() {
   case cty.String:return map[string]any{"cty":"string","value":v.AsString()}
   case cty.Bool:return map[string]any{"cty":"bool","value":v.True()}
   case cty.Number:return map[string]any{"cty":"number","value":v.AsBigFloat().Text('g',-1)}
   }
   panic("unexpected parser cty literal")
  case *Operation:
   for name,op:=range map[string]*Operation{"or":OpLogicalOr,"and":OpLogicalAnd,"not":OpLogicalNot,"equal":OpEqual,"notEqual":OpNotEqual,"greater":OpGreaterThan,"greaterEqual":OpGreaterThanOrEqual,"less":OpLessThan,"lessEqual":OpLessThanOrEqual,"add":OpAdd,"subtract":OpSubtract,"multiply":OpMultiply,"divide":OpDivide,"modulo":OpModulo,"negate":OpNegate}{if v==op{return name}}
   panic("unknown operation")
  }
 }
 switch value.Kind() {
 case reflect.Pointer:if value.IsNil(){return nil};return reyloOracleValue(value.Elem())
 case reflect.Struct:
  result:=map[string]any{"goType":value.Type().Name()}
  for i:=0;i<value.NumField();i++ {field:=value.Type().Field(i);if !field.IsExported(){continue};result[field.Name]=reyloOracleValue(value.Field(i))}
  if value.CanAddr() && value.Addr().CanInterface() { if node,ok:=value.Addr().Interface().(interface{Range() hcl.Range});ok{result["ComputedRange"]=reyloOracleValue(reflect.ValueOf(node.Range()))} }
  return result
 case reflect.Map:
  result:=map[string]any{}
  iter:=value.MapRange();for iter.Next(){result[fmt.Sprint(iter.Key().Interface())]=reyloOracleValue(iter.Value())};return result
 case reflect.Array,reflect.Slice:
  result:=[]any{};for i:=0;i<value.Len();i++{result=append(result,reyloOracleValue(value.Index(i)))};return result
 case reflect.String:return value.String()
 case reflect.Bool:return value.Bool()
 case reflect.Int,reflect.Int8,reflect.Int16,reflect.Int32,reflect.Int64:return value.Int()
 case reflect.Uint,reflect.Uint8,reflect.Uint16,reflect.Uint32,reflect.Uint64:return value.Uint()
 default:panic(fmt.Sprintf("unsupported oracle reflection %s",value.Type()))
 }
}

func reyloOracleRecord(api,name string,src []byte,filename string,start hcl.Pos,result any,diags hcl.Diagnostics,extra any) {
 fixture:=map[string]any{"api":api,"name":name,"source":src,"filename":filename,"start":start,"result":reyloOracleValue(reflect.ValueOf(result)),"diagnostics":reyloOracleValue(reflect.ValueOf(diags)),"extra":extra}
 data,err:=json.Marshal(fixture);if err!=nil{panic(err)}
 reyloOracleMutex.Lock();defer reyloOracleMutex.Unlock()
 f,err:=os.OpenFile(os.Getenv("HCL_ORACLE_OUTPUT"),os.O_CREATE|os.O_APPEND|os.O_WRONLY,0644);if err!=nil{panic(err)}
 defer f.Close();if _,err=f.Write(append(data,'\n'));err!=nil{panic(err)}
}
